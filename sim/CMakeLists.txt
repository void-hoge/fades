find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(verilator REQUIRED HINTS $ENV{VERILATOR_ROOT})
find_program(IVERILOG_EXECUTABLE NAMES iverilog REQUIRED)
find_program(VVP_EXECUTABLE NAMES vvp REQUIRED)
find_program(YOSYS_EXECUTABLE NAMES yosys REQUIRED)
include("${PROJECT_SOURCE_DIR}/cmake/RtlTests.cmake")
set(FADES_DATASET "${CMAKE_CURRENT_SOURCE_DIR}/data/17puz49158.txt" CACHE FILEPATH "Test dataset")
set(FADES_BASELINE "" CACHE FILEPATH "Full core-test CSV baseline for regression")
foreach(suite IN ITEMS deduction_engine branch_generator segment_rules fades_core fades fades_axi async_queue reference)
    add_subdirectory(${suite})
endforeach()

foreach(top IN ITEMS fades_core fades fades_axi fades_kv260)
    set(suite "${top}")
    if(top STREQUAL "fades_core")
        set(sources ${FADES_CORE_RTL})
        set(params -GSEGMENT_RULES=${FADES_SEGMENT_RULES})
    elseif(top STREQUAL "fades")
        set(sources ${FADES_STREAM_RTL})
        set(params -GCORES=${FADES_CORES} -GSEGMENT_RULES=${FADES_SEGMENT_RULES})
    else()
        set(sources ${FADES_AXI_RTL})
        set(params -GCORES=${FADES_CORES} -GSEGMENT_RULES=${FADES_SEGMENT_RULES})
        if(top STREQUAL "fades_kv260")
            # Board wrapper checks belong with the AXI integration boundary.
            set(suite fades_axi)
            list(APPEND sources "${PROJECT_SOURCE_DIR}/fpga/kv260/rtl/fades_kv260.sv")
        endif()
    endif()
    add_test(NAME lint-${top} COMMAND "${VERILATOR_BIN}" --lint-only --language 1800-2017
        -Wall --top-module ${top} ${params} ${sources})
    set_tests_properties(lint-${top} PROPERTIES LABELS "${suite}" TIMEOUT 120)
endforeach()
add_custom_target(lint COMMAND "${CMAKE_CTEST_COMMAND}" --output-on-failure -R "^lint-"
    WORKING_DIRECTORY "${PROJECT_BINARY_DIR}" USES_TERMINAL VERBATIM)
add_custom_target(formal COMMAND "${CMAKE_CTEST_COMMAND}" --output-on-failure -R "^formal-"
    WORKING_DIRECTORY "${PROJECT_BINARY_DIR}" USES_TERMINAL VERBATIM)
